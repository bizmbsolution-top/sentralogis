### 5. Database Schema (Mental Draft)

```sql
-- Driver Attendance
CREATE TABLE driver_attendance (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  driver_id UUID REFERENCES md_drivers(id),
  date DATE NOT NULL,
  check_in TIMESTAMPTZ,
  check_out TIMESTAMPTZ,
  status TEXT DEFAULT 'present',
  location_lat FLOAT,
  location_lng FLOAT,
  photo_url TEXT,
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(driver_id, date)
);

-- Fleet Inspections
CREATE TABLE fleet_inspections (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  fleet_id UUID REFERENCES md_fleets(id),
  driver_id UUID REFERENCES md_drivers(id),
  inspection_date DATE NOT NULL,
  checklist_data JSONB,
  photos JSONB,
  overall_status TEXT DEFAULT 'pass',
  notes TEXT,
  inspector_id UUID REFERENCES profiles(id),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Add to job_orders
ALTER TABLE job_orders ADD COLUMN IF NOT EXISTS start_odometer INTEGER;
ALTER TABLE job_orders ADD COLUMN IF NOT EXISTS end_odometer INTEGER;
ALTER TABLE job_orders ADD COLUMN IF NOT EXISTS distance_km_actual FLOAT;
```
