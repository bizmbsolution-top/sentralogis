-- View: Current Stock per Customer per Warehouse
CREATE VIEW v_customer_stock AS
SELECT
    i.customer_id,
    c.customer_name,
    i.sku,
    ps.product_name,
    w.warehouse_id,
    w.warehouse_name,
    a.area_type,
    i.lot_number,
    i.expiry_date,
    i.status,
    COUNT(*) AS item_count,
    SUM(i.current_quantity) AS total_quantity,
    lb.bin_id AS location
FROM inventory i
JOIN customer c ON i.customer_id = c.customer_id
JOIN product_sku ps ON i.sku = ps.sku
JOIN location_bin lb ON i.current_bin_id = lb.bin_id
JOIN zone z ON lb.zone_id = z.zone_id
JOIN area a ON z.area_id = a.area_id
JOIN warehouse w ON a.warehouse_id = w.warehouse_id
WHERE i.status IN ('AVAILABLE', 'RESERVED')
GROUP BY i.customer_id, c.customer_name, i.sku, ps.product_name,
         w.warehouse_id, w.warehouse_name, a.area_type, i.lot_number,
         i.expiry_date, i.status, lb.bin_id;

-- View: Warehouse Utilization
CREATE VIEW v_warehouse_utilization AS
SELECT
    w.warehouse_id,
    w.warehouse_name,
    a.area_id,
    a.area_name,
    a.area_type,
    a.total_capacity,
    a.uom_capacity,
    COUNT(DISTINCT i.inventory_id) AS occupied_units,
    SUM(i.current_quantity * ps.weight_kg) AS total_weight_kg,
    (COUNT(DISTINCT i.inventory_id)::DECIMAL / NULLIF(a.total_capacity, 0)) * 100 AS utilization_pct
FROM warehouse w
JOIN area a ON w.warehouse_id = a.warehouse_id
JOIN zone z ON a.area_id = z.area_id
JOIN location_bin lb ON z.zone_id = lb.zone_id
LEFT JOIN inventory i ON lb.bin_id = i.current_bin_id AND i.status IN ('AVAILABLE', 'RESERVED')
LEFT JOIN product_sku ps ON i.sku = ps.sku
GROUP BY w.warehouse_id, w.warehouse_name, a.area_id, a.area_name, a.area_type,
         a.total_capacity, a.uom_capacity;

-- Function: Get FEFO Pick Suggestion
CREATE OR REPLACE FUNCTION get_fefo_pick(p_sku VARCHAR, p_quantity DECIMAL)
RETURNS TABLE(
    inventory_id VARCHAR,
    lot_number VARCHAR,
    expiry_date DATE,
    current_bin_id VARCHAR,
    available_quantity DECIMAL
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        i.inventory_id,
        i.lot_number,
        i.expiry_date,
        i.current_bin_id,
        i.current_quantity
    FROM inventory i
    WHERE i.sku = p_sku
      AND i.status = 'AVAILABLE'
    ORDER BY i.expiry_date ASC NULLS LAST, i.receipt_date ASC
    LIMIT (SELECT CEIL(p_quantity)::INTEGER);
END;
$$ LANGUAGE plpgsql;

-- Function: Calculate Monthly Storage Charge
CREATE OR REPLACE FUNCTION calculate_storage_charge(
    p_contract_id VARCHAR,
    p_period_start DATE,
    p_period_end DATE
) RETURNS DECIMAL AS $$
DECLARE
    avg_daily_pallet DECIMAL;
    committed_space DECIMAL;
    overflow_space DECIMAL;
    fixed_rate DECIMAL;
    overflow_rate DECIMAL;
    multiplier DECIMAL;
    total_charge DECIMAL := 0;
BEGIN
    -- Get average daily pallet from snapshot
    SELECT AVG(total_pallet) INTO avg_daily_pallet
    FROM daily_stock_snapshot
    WHERE contract_id = p_contract_id
      AND snapshot_date BETWEEN p_period_start AND p_period_end;

    -- Get contract details
    SELECT sc.committed_space, sc.max_overflow, sc.overflow_rate_multiplier
    INTO committed_space, overflow_space, multiplier
    FROM storage_contract sc
    WHERE sc.contract_id = p_contract_id;

    -- Get rate
    SELECT rate_value INTO fixed_rate
    FROM billing_rate
    WHERE contract_id = p_contract_id
      AND charge_code = 'STR-FIX'
      AND status = 'ACTIVE';

    -- Calculate overflow
    IF avg_daily_pallet > committed_space THEN
        overflow_space := LEAST(avg_daily_pallet - committed_space, COALESCE(overflow_space, 0));
        total_charge := (committed_space * fixed_rate) + (overflow_space * fixed_rate * multiplier);
    ELSE
        total_charge := avg_daily_pallet * fixed_rate;
    END IF;

    RETURN total_charge;
END;
$$ LANGUAGE plpgsql;

-- Trigger: Update location_bin status after inventory change
CREATE OR REPLACE FUNCTION update_bin_status()
RETURNS TRIGGER AS $$
BEGIN
    -- If inventory moved out, check if bin is empty
    IF OLD.current_bin_id IS NOT NULL THEN
        IF NOT EXISTS (
            SELECT 1 FROM inventory
            WHERE current_bin_id = OLD.current_bin_id
              AND status IN ('AVAILABLE', 'RESERVED', 'QUARANTINE')
              AND inventory_id != OLD.inventory_id
        ) THEN
            UPDATE location_bin SET status = 'EMPTY', current_weight_kg = 0
            WHERE bin_id = OLD.current_bin_id;
        END IF;
    END IF;

    -- If inventory moved in, mark bin occupied
    IF NEW.current_bin_id IS NOT NULL THEN
        UPDATE location_bin SET status = 'OCCUPIED'
        WHERE bin_id = NEW.current_bin_id;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_update_bin_status
AFTER UPDATE OF current_bin_id ON inventory
FOR EACH ROW
EXECUTE FUNCTION update_bin_status();

-- Trigger: Log all inventory movements
CREATE OR REPLACE FUNCTION log_inventory_movement()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.current_bin_id IS DISTINCT FROM NEW.current_bin_id THEN
        INSERT INTO movement_log (
            movement_id, inventory_id, movement_type,
            from_bin_id, to_bin_id, from_status, to_status, quantity
        ) VALUES (
            'MV-' || TO_CHAR(NOW(), 'YYYYMMDD') || '-' || NEXTVAL('movement_seq'),
            NEW.inventory_id,
            CASE
                WHEN OLD.current_bin_id IS NULL THEN 'PUTAWAY'
                WHEN NEW.current_bin_id IS NULL THEN 'REMOVAL'
                ELSE 'RELOCATION'
            END,
            OLD.current_bin_id,
            NEW.current_bin_id,
            OLD.status,
            NEW.status,
            NEW.current_quantity
        );
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_log_movement
AFTER UPDATE ON inventory
FOR EACH ROW
EXECUTE FUNCTION log_inventory_movement();
