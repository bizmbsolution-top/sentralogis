## Task 3: API — `/api/cron/sla-snapshot`
**Estimasi: 15 menit**

Buat file: `app/api/cron/sla-snapshot/route.ts`

```ts
import { NextResponse } from 'next/server';
import { createClient } from '@/lib/supabase/server';

export async function GET(req: Request) {
  // 1. Verify cron secret
  const authHeader = req.headers.get('authorization');
  if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  // 2. Fetch all active tenants
  const supabase = createClient();
  const { data: tenants } = await supabase
    .from('tenants')
    .select('id')
    .eq('is_active', true);

  // 3. For each tenant, call capture_sla_snapshot
  const results = [];
  for (const tenant of tenants || []) {
    await supabase.rpc('capture_sla_snapshot', { p_tenant_id: tenant.id });
    results.push(tenant.id);
  }

  return NextResponse.json({ 
    captured: results.length,
    date: new Date().toISOString().split('T')[0]
  });
}
```
