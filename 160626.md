# Panduan Eksekusi Perbaikan SLA - 16 Juni 2026

Dokumen ini berisi panduan dan skrip SQL untuk memperbaiki sistem pelacakan SLA (Work Order Draft → Submit ke SBU) yang mengalami error akibat kolom `wo.submitted_at` tidak ditemukan pada database produksi.

---

## 📋 Masalah yang Diselesaikan
1. Error database PostgreSQL: `column wo.submitted_at does not exist` saat menekan tombol Megaphone (Trigger SLA Escalation) atau ketika API `/api/admin/sla-escalation` berjalan secara berkala.
2. Penambahan kolom pelacakan waktu transisi SLA secara riil pada tabel `work_orders` agar proses rekam waktu berjalan akurat.

---

## 🛠️ Langkah-langkah Eksekusi (16 Juni 2026)

### Langkah 1: Eksekusi SQL di Supabase SQL Editor
Buka **Supabase Dashboard → SQL Editor → New Query**, tempel kode SQL berikut, lalu klik **Run**:

```sql
-- =========================================================================
-- 1. TAMBAH KOLOM SLA TIMESTAMPS PADA TABEL WORK_ORDERS
-- =========================================================================
ALTER TABLE public.work_orders ADD COLUMN IF NOT EXISTS submitted_at TIMESTAMPTZ;
ALTER TABLE public.work_orders ADD COLUMN IF NOT EXISTS assigned_at TIMESTAMPTZ;
ALTER TABLE public.work_orders ADD COLUMN IF NOT EXISTS invoiced_at TIMESTAMPTZ;
ALTER TABLE public.work_orders ADD COLUMN IF NOT EXISTS paid_at TIMESTAMPTZ;

-- =========================================================================
-- 2. BUAT TRIGGER UNTUK OTOMATISASI PENGISIAN TIMESTAMPS
-- =========================================================================
CREATE OR REPLACE FUNCTION public.update_work_order_sla_timestamps()
RETURNS TRIGGER AS $$
BEGIN
  -- SLA 1 -> 2: Draft ke Submit
  IF (UPPER(NEW.status) NOT IN ('DRAFT', 'REJECTED') AND (OLD.status IS NULL OR UPPER(OLD.status) = 'DRAFT')) THEN
    NEW.submitted_at := NOW();
  END IF;

  -- SLA 2 -> 3: Assigned ke SBU / Sedang Berjalan
  IF (UPPER(NEW.status) IN ('ASSIGNED', 'IN_PROGRESS', 'ON_JOURNEY') AND (OLD.status IS NULL OR UPPER(OLD.status) NOT IN ('ASSIGNED', 'IN_PROGRESS', 'ON_JOURNEY'))) THEN
    NEW.assigned_at := NOW();
  END IF;

  -- SLA 3 -> 4: Pekerjaan Selesai
  IF (UPPER(NEW.status) IN ('COMPLETED', 'READY_FOR_BILLING') AND (OLD.status IS NULL OR UPPER(OLD.status) NOT IN ('COMPLETED', 'READY_FOR_BILLING'))) THEN
    NEW.completed_at := NOW();
  END IF;

  -- SLA 4 -> 5: Tagihan Terbit (Invoiced)
  IF (UPPER(NEW.status) IN ('INVOICED') AND (OLD.status IS NULL OR UPPER(OLD.status) != 'INVOICED')) THEN
    NEW.invoiced_at := NOW();
  END IF;

  -- SLA 5 -> 6: Lunas (Paid)
  IF (UPPER(NEW.status) IN ('PAID') AND (OLD.status IS NULL OR UPPER(OLD.status) != 'PAID')) THEN
    NEW.paid_at := NOW();
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Pasang Trigger pada Tabel work_orders
DROP TRIGGER IF EXISTS trg_update_wo_sla_timestamps ON public.work_orders;
CREATE TRIGGER trg_update_wo_sla_timestamps
  BEFORE UPDATE OF status ON public.work_orders
  FOR EACH ROW
  EXECUTE FUNCTION public.update_work_order_sla_timestamps();

-- =========================================================================
-- 3. MIGRASI DATA LAMA (BACKFILL)
-- =========================================================================
UPDATE public.work_orders
SET 
  submitted_at = CASE WHEN UPPER(status) NOT IN ('DRAFT', 'REJECTED') THEN created_at ELSE NULL END,
  assigned_at = CASE WHEN UPPER(status) IN ('ASSIGNED', 'IN_PROGRESS', 'ON_JOURNEY', 'COMPLETED', 'READY_FOR_BILLING', 'INVOICED', 'PAID') THEN updated_at ELSE NULL END,
  completed_at = CASE WHEN UPPER(status) IN ('COMPLETED', 'READY_FOR_BILLING', 'INVOICED', 'PAID') THEN completed_at ELSE NULL END,
  invoiced_at = CASE WHEN UPPER(status) IN ('INVOICED', 'PAID') THEN updated_at ELSE NULL END,
  paid_at = CASE WHEN UPPER(status) IN ('PAID') THEN updated_at ELSE NULL END
WHERE tenant_id IS NOT NULL;

-- =========================================================================
-- 4. PERBARUI FUNGSI DETEKSI SLA BREACH (get_active_sla_breaches)
-- =========================================================================
CREATE OR REPLACE FUNCTION public.get_active_sla_breaches(p_tenant_id UUID)
RETURNS TABLE (
  wo_id UUID,
  jo_id UUID,
  breach_type TEXT,
  stage TEXT,
  overdue_minutes INTEGER,
  tenant_id UUID,
  wo_number TEXT,
  jo_number TEXT,
  details TEXT
) AS $$
BEGIN
  RETURN QUERY
  -- SLA 1: WO masih DRAFT setelah 30 menit dari created_at
  SELECT 
    wo.id AS wo_id,
    NULL::UUID AS jo_id,
    'SLA 1'::TEXT AS breach_type,
    'SLA 1'::TEXT AS stage,
    (EXTRACT(EPOCH FROM (NOW() - wo.created_at))/60 - 30)::INTEGER AS overdue_minutes,
    wo.tenant_id AS tenant_id,
    wo.wo_number AS wo_number,
    '—'::TEXT AS jo_number,
    'Work Order masih Draft lebih dari 30 menit'::TEXT AS details
  FROM public.work_orders wo
  WHERE wo.tenant_id = p_tenant_id
    AND UPPER(wo.status) = 'DRAFT'
    AND wo.created_at < (NOW() - INTERVAL '30 minutes')

  UNION ALL

  -- SLA 2: WO sudah di-submit tapi belum diassign/diproses setelah 60 menit
  SELECT 
    wo.id AS wo_id,
    NULL::UUID AS jo_id,
    'SLA 2'::TEXT AS breach_type,
    'SLA 2'::TEXT AS stage,
    (EXTRACT(EPOCH FROM (NOW() - COALESCE(wo.submitted_at, wo.created_at)))/60 - 60)::INTEGER AS overdue_minutes,
    wo.tenant_id AS tenant_id,
    wo.wo_number AS wo_number,
    '—'::TEXT AS jo_number,
    'Work Order belum diassign ke SBU setelah 60 menit'::TEXT AS details
  FROM public.work_orders wo
  WHERE wo.tenant_id = p_tenant_id
    AND UPPER(wo.status) IN ('PENDING', 'NEED_ASSIGNMENT', 'PENDING_APPROVAL')
    AND COALESCE(wo.submitted_at, wo.created_at) < (NOW() - INTERVAL '60 minutes')

  UNION ALL

  -- SLA 3: JO sudah Selesai tapi Dokumen atau Cost belum lengkap setelah 3 hari (4320 menit)
  SELECT 
    wo.id AS wo_id,
    jo.id AS jo_id,
    'SLA 3'::TEXT AS breach_type,
    'SLA 3'::TEXT AS stage,
    (EXTRACT(EPOCH FROM (NOW() - jo.actual_end))/60 - 4320)::INTEGER AS overdue_minutes,
    wo.tenant_id AS tenant_id,
    wo.wo_number AS wo_number,
    jo.jo_number AS jo_number,
    CASE 
      WHEN NOT jo.is_doc_finished AND NOT jo.is_cost_finished THEN 'Dokumen dan biaya JO belum lengkap'
      WHEN NOT jo.is_doc_finished THEN 'Dokumen JO belum diunggah/selesai'
      ELSE 'Audit biaya JO belum selesai'
    END::TEXT AS details
  FROM public.job_orders jo
  JOIN public.work_orders wo ON jo.work_order_id = wo.id
  WHERE wo.tenant_id = p_tenant_id
    AND UPPER(jo.status) IN ('COMPLETED', 'PEKERJAAN SELESAI', 'SELESAI', 'DONE')
    AND (jo.is_doc_finished = FALSE OR jo.is_cost_finished = FALSE)
    AND jo.actual_end < (NOW() - INTERVAL '3 days')

  UNION ALL

  -- SLA 4: Dokumen & Biaya lengkap tapi belum dibuatkan Invoice ke customer setelah 1 hari (1440 menit)
  SELECT 
    wo.id AS wo_id,
    NULL::UUID AS jo_id,
    'SLA 4'::TEXT AS breach_type,
    'SLA 4'::TEXT AS stage,
    (EXTRACT(EPOCH FROM (NOW() - COALESCE(wo.completed_at, wo.updated_at)))/60 - 1440)::INTEGER AS overdue_minutes,
    wo.tenant_id AS tenant_id,
    wo.wo_number AS wo_number,
    '—'::TEXT AS jo_number,
    'Work Order siap billing belum dibuatkan Invoice selama lebih dari 1 hari'::TEXT AS details
  FROM public.work_orders wo
  WHERE wo.tenant_id = p_tenant_id
    AND UPPER(wo.status) IN ('COMPLETED', 'READY_FOR_BILLING')
    AND NOT EXISTS (
      SELECT 1 FROM public.job_orders jo 
      WHERE jo.work_order_id = wo.id 
        AND (jo.is_doc_finished = FALSE OR jo.is_cost_finished = FALSE)
    )
    AND COALESCE(wo.completed_at, wo.updated_at) < (NOW() - INTERVAL '1 day');
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

## 🧪 Langkah 2: Cara Verifikasi

1. **Uji Coba Pemicu Manual**:
   * Buka Dashboard Operations di browser Anda: `/hq/ops-dashboard`.
   * Klik tombol **Megaphone** (Trigger SLA Escalation).
   * Verifikasi bahwa toast notifikasi hijau muncul (misalnya: *X escalation(s), Y notification(s) sent*) tanpa menampilkan error API.
2. **Uji Coba Pembuatan WO**:
   * Buat WO baru berstatus Draft.
   * Submit WO tersebut ke SBU terkait.
   * Periksa di tabel `work_orders` bahwa kolom `submitted_at` terisi dengan waktu saat Anda menekan tombol submit secara otomatis.
